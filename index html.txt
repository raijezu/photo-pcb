    // ===== Comunicação com o servidor =====
    // Dentro do Apps Script: usa google.script.run normalmente.
    // Fora dele (página num site próprio, para a câmera ao vivo funcionar no iPhone): as MESMAS
    // chamadas google.script.run.xxx() vão pela internet para a API (doPost do code.gs).
    const API_URL = 'https://script.google.com/macros/s/AKfycbzY-R7JYPLGzxs7PDlI1xvMGxj5goOSWT7VStHPd_z_rZ3ECoJ6JxFUZlQFbkJQIM2ScA/exec'; // a URL do web app publicado, termina em /exec
    const CHAVE_API = 'NykRSZf_XpRYdJGDgLl7Y7R_';                 // a mesma do code.gs

    if (!(window.google && google.script && google.script.run)) {
      window.google = { script: { get run() { return chamadaPelaApi(); } } };
    }

    function chamadaPelaApi() {
      let aoSucesso = function() {};
      let aoFalhar = function(e) { console.error(e); };
      const base = {
        withSuccessHandler: function(f) { aoSucesso = f; return chamada; },
        withFailureHandler: function(f) { aoFalhar = f; return chamada; }
      };
      const chamada = new Proxy(base, {
        get: function(alvo, nome) {
          if (nome in alvo) return alvo[nome];
          return function() {
            if (API_URL.indexOf('/exec') < 0) {
              aoFalhar(new Error('Configure a API_URL no index.html (URL do web app que termina em /exec).'));
              return;
            }
            fetch(API_URL, {
              method: 'POST',
              headers: { 'Content-Type': 'text/plain;charset=utf-8' }, // evita bloqueio de CORS
              body: JSON.stringify({ acao: nome, args: Array.prototype.slice.call(arguments), chave: CHAVE_API })
            })
              .then(function(r) { return r.json(); })
              .then(function(d) { if (d.ok) aoSucesso(d.resultado); else aoFalhar(new Error(d.erro)); })
              .catch(function(e) { aoFalhar(new Error('Sem conexão com o servidor (' + e.message + ').')); });
          };
        }
      });
      return chamada;
    }

    //  - na tela: cópia reduzida (LIMITE_TELA) com os MESMOS ajustes da foto final;
    //  - foto final: recorte em resolução cheia -> reduzida para LIMITE_SAIDA -> ajustes -> JPEG.
    // (Antes o Cropper reduzia a foto inteira para 1600 px ANTES de recortar e depois ampliava o
    //  recorte, por isso a foto salva ficava borrada/diferente do que se via no editor.)

    function carregarImagem(src) {
      return new Promise(function(ok, falha) {
        const img = new Image();
        img.onload = function() { ok(img); };
        img.onerror = function() { falha(new Error('Não foi possível abrir a imagem.')); };
        img.src = src;
      });
    }

    // Reduz com boa qualidade (em etapas de metade) até caber em "limite" px no lado maior
    function redimensionar(fonte, limite) {
      let w = fonte.naturalWidth || fonte.width, h = fonte.naturalHeight || fonte.height;
      const escala = Math.min(1, limite / Math.max(w, h));
      const alvoW = Math.max(1, Math.round(w * escala)), alvoH = Math.max(1, Math.round(h * escala));
      let atual = fonte;
      while (w / 2 >= alvoW && h / 2 >= alvoH) {
        const c = document.createElement('canvas');
        c.width = Math.round(w / 2); c.height = Math.round(h / 2);
        const x = c.getContext('2d'); x.imageSmoothingQuality = 'high';
        x.drawImage(atual, 0, 0, c.width, c.height);
        atual = c; w = c.width; h = c.height;
      }
      const final = document.createElement('canvas');
      final.width = alvoW; final.height = alvoH;
      const ctx = final.getContext('2d'); ctx.imageSmoothingQuality = 'high';
      ctx.drawImage(atual, 0, 0, alvoW, alvoH);
      return final;
    }

    function canvasParaUrl(canvas, qualidade) {
      return new Promise(function(ok) {
        canvas.toBlob(function(blob) { ok(URL.createObjectURL(blob)); }, 'image/jpeg', qualidade);
      });
    }

    // Conta de cores ÚNICA (tela e foto final usam a mesma):
    // saída = m * entrada + off   (valores de 0 a 255; o resultado é limitado a 0..255)
    function coeficientesCor(b, c, s) {
      const add = b * 2.55;
      const cc = c * 2.55;
      const fc = (259 * (cc + 255)) / (255 * (259 - cc));   // contraste
      const fs = 1 + s / 100;                                // saturação
      const pr = [0.299, 0.587, 0.114];
      const m = [0, 1, 2].map(function(i) {
        return [0, 1, 2].map(function(j) {
          return fc * ((i === j ? fs : 0) + (1 - fs) * pr[j]);
        });
      });
      return { m: m, off: fc * (add - 128) + 128 };
    }

    function fatorNitidez(n) { return n / 100 * 1.2; }

    // Aplica brilho, contraste, saturação e nitidez no canvas
    function processarCanvas(canvas, b, c, s, nit) {
      if (b || c || s) {
        const ctx = canvas.getContext('2d');
        const dados = ctx.getImageData(0, 0, canvas.width, canvas.height);
        const px = dados.data;
        const co = coeficientesCor(b, c, s);
        const m = co.m, off = co.off;
        const lim = v => v < 0 ? 0 : v > 255 ? 255 : v;
        for (let i = 0; i < px.length; i += 4) {
          const r = px[i], g = px[i + 1], bl = px[i + 2];
          px[i]     = lim(m[0][0] * r + m[0][1] * g + m[0][2] * bl + off);
          px[i + 1] = lim(m[1][0] * r + m[1][1] * g + m[1][2] * bl + off);
          px[i + 2] = lim(m[2][0] * r + m[2][1] * g + m[2][2] * bl + off);
        }
        ctx.putImageData(dados, 0, 0);
      }
      if (nit) aplicarNitidez(canvas, fatorNitidez(nit));
    }

    // Nitidez: realce 3x3 (a = intensidade); as bordas da imagem ficam como estão
    function aplicarNitidez(canvas, a) {
      const w = canvas.width, h = canvas.height;
      const ctx = canvas.getContext('2d');
      const origem = ctx.getImageData(0, 0, w, h);
      const destino = ctx.createImageData(w, h);
      const s = origem.data, d = destino.data;
      const lim = v => v < 0 ? 0 : v > 255 ? 255 : v;
      d.set(s);
      for (let y = 1; y < h - 1; y++) {
        for (let x = 1; x < w - 1; x++) {
          const i = (y * w + x) * 4;
          for (let c = 0; c < 3; c++) {
            const centro = s[i + c];
            const vizinhos = s[i - 4 + c] + s[i + 4 + c] + s[i - w * 4 + c] + s[i + w * 4 + c];
            d[i + c] = lim(centro + a * (4 * centro - vizinhos));
          }
        }
      }
      ctx.putImageData(destino, 0, 0);
    }

    // Gera a foto final. É o MESMO caminho da "Prévia do resultado" e do salvamento.
    function gerarResultado() {
      if (!cropper || !baseCanvas) return null;
      const a = lerAjustes();
      const exibida = cropper.image;
      let recorte;
      cropper.image = baseCanvas; // recorta da foto em resolução cheia (sem reduzir antes)
      try {
        recorte = cropper.getCroppedCanvas({ fillColor: '#fff', imageSmoothingQuality: 'high' });
      } finally {
        cropper.image = exibida;
      }
      if (!recorte) return null;
      const saida = redimensionar(recorte, LIMITE_SAIDA);
      processarCanvas(saida, a.b, a.c, a.s, a.n);
      return { canvas: saida, dataUrl: saida.toDataURL('image/jpeg', a.q / 100) };
    }

    // Código(s): vários códigos no mesmo pacote. A cada 5 dígitos o sistema coloca " / "
    // sozinho ao continuar digitando (ex.: 48683 / 48682). Qualquer não-número (espaço, / , ; -)
    // encerra um código com menos de 5 dígitos.
    const TAM_CODIGO = 5;
    function formatarCodigos(bruto, manterSeparadorFinal) {
      const grupos = [];
      bruto.split(/[^0-9]+/).forEach(function(parte) {
        for (let i = 0; i < parte.length; i += TAM_CODIGO) grupos.push(parte.slice(i, i + TAM_CODIGO));
      });
      let texto = grupos.join(' / ');
      if (manterSeparadorFinal && grupos.length && /[^0-9]$/.test(bruto)) texto += ' / ';
      return texto;
    }

      // dígito digitado logo após um código completo ganha o " / " automático
      campoCodigo.value = formatarCodigos(campoCodigo.value, !apagando);

    // ===== Leitura do QR da etiqueta =====
    // Formato: C47142Q5000.000U0013593571 -> C = código, Q = quantidade por pacote
    function parsearEtiqueta(texto) {
      const m = /C(\d+)Q(\d+(?:[.,]\d+)?)/i.exec((texto || '').trim());
      if (!m) return null;
      return { codigo: m[1], quantPacote: Math.round(parseFloat(m[2].replace(',', '.'))) };
    }

        return detector.detect(teste).then(function() { detectorNativo = detector; });
      }).catch(function() { usarLeitorAntigo(); })
        .then(function() { preparando = null; });
      return preparando;
    }

    // Tenta ler um canvas; devolve o texto ou null
    function decodificar(canvas) {
      if (detectorNativo) {
        return detectorNativo.detect(canvas)
          .then(function(r) { return r.length ? r[0].rawValue : null; })
          .catch(function() { usarLeitorAntigo(); return decodificar(canvas); });
      }
      return new Promise(function(ok) {
        canvas.toBlob(function(blob) {
          const arq = new File([blob], 'quadro.jpg', { type: 'image/jpeg' });
          leitorArquivo.scanFileV2(arq, false)
            .then(function(r) { ok(r && r.decodedText ? r.decodedText : null); })
            .catch(function() { ok(null); });
        }, 'image/jpeg', 0.95);
      });
    }

    // Espera a câmera mandar imagem de verdade (não preta) por até "ms"
    function esperarImagem(sessao, ms) {
      const amostra = document.createElement('canvas');
      amostra.width = 32; amostra.height = 32;
      const ctx = amostra.getContext('2d');
      const inicio = Date.now();
      return new Promise(function(ok) {
        (function checar() {
          if (sessao !== sessaoQR) { ok(false); return; }
          if (videoQR.videoWidth > 0 && videoQR.readyState >= 2) {
            try {
              ctx.drawImage(videoQR, 0, 0, 32, 32);
              const d = ctx.getImageData(0, 0, 32, 32).data;
              let max = 0;
              for (let i = 0; i < d.length; i += 4) max = Math.max(max, d[i], d[i + 1], d[i + 2]);
              if (max > 10) { ok(true); return; }
            } catch (e) {}
          }
          if (Date.now() - inicio > ms) { ok(false); return; }
          setTimeout(checar, 250);
        })();
      });
    }

    function nomeErroCamera(err) {
      const n = (err && err.name) || '';
      if (n === 'NotAllowedError' || n === 'SecurityError' || n === 'PermissionDeniedError')
        return 'O acesso à câmera está bloqueado. Toque no cadeado ao lado do endereço do site, permita a câmera e depois toque em "Tentar de novo".';
      if (n === 'NotReadableError' || n === 'TrackStartError')
        return 'A câmera está sendo usada por outro aplicativo ou aba. Feche o outro uso e toque em "Tentar de novo".';
      if (n === 'NotFoundError' || n === 'DevicesNotFoundError')
        return 'Nenhuma câmera foi encontrada neste aparelho.';
      return 'Não foi possível abrir a câmera' + (n ? ' (' + n + ')' : '') + '.';
    }

      });

    // ----- Achar o código na imagem -----
    // O leitor de Data Matrix (ZXing) só lê quando o código está QUASE NO CENTRO do recorte.
    // Então procuramos áreas quadradas cheias de bordas finas (horizontais e verticais), que é a
    // "cara" de um QR/Data Matrix, e recortamos centrado em cada uma.
    function candidatosCodigo(fonte, W, H, quantos) {
      const s = Math.min(1, 480 / Math.max(W, H));
      const w = Math.max(8, Math.round(W * s)), h = Math.max(8, Math.round(H * s));
      const c = document.createElement('canvas'); c.width = w; c.height = h;
      const ctx = c.getContext('2d');
      ctx.drawImage(fonte, 0, 0, w, h);
      const d = ctx.getImageData(0, 0, w, h).data;
      const g = new Float32Array(w * h);
      for (let i = 0, j = 0; j < g.length; i += 4, j++) g[j] = d[i] * 0.299 + d[i + 1] * 0.587 + d[i + 2] * 0.114;
      const L = w + 1;
      const IH = new Float64Array(L * (h + 1)), IV = new Float64Array(L * (h + 1));
      for (let y = 0; y < h; y++) {
        let sh = 0, sv = 0;
        for (let x = 0; x < w; x++) {
          const k = y * w + x;
          if (x + 1 < w && Math.abs(g[k + 1] - g[k]) > 35) sh++;
          if (y + 1 < h && Math.abs(g[k + w] - g[k]) > 35) sv++;
          const o = (y + 1) * L + (x + 1);
          IH[o] = IH[o - L] + sh;
          IV[o] = IV[o - L] + sv;
        }
      }
      function soma(I, x0, y0, x1, y1) { return I[y1 * L + x1] - I[y0 * L + x1] - I[y1 * L + x0] + I[y0 * L + x0]; }
      const achados = [];
      [10, 14, 20, 28, 40, 56, 80, 110, 150].forEach(function(ws) {
        if (ws >= Math.min(w, h)) return;
        const area = ws * ws, passo = Math.max(1, Math.round(ws / 8));
        for (let y0 = 0; y0 + ws <= h; y0 += passo) {
          for (let x0 = 0; x0 + ws <= w; x0 += passo) {
            const dens = Math.min(soma(IH, x0, y0, x0 + ws, y0 + ws), soma(IV, x0, y0, x0 + ws, y0 + ws)) / area;
            // janela do tamanho do código ganha (menor: mesma densidade; maior: densidade cai)
            if (dens > 0.12) achados.push({ x: x0 + ws / 2, y: y0 + ws / 2, ws: ws, sc: dens * Math.sqrt(ws) });
          }
        }
      });
      achados.sort(function(a, b) { return b.sc - a.sc; });
      const escolhidos = [];
      for (let i = 0; i < achados.length && escolhidos.length < quantos; i++) {
        const a = achados[i];
        if (escolhidos.some(function(e) { return Math.hypot(e.x - a.x, e.y - a.y) < Math.max(e.ws, a.ws) * 0.75; })) continue;
        escolhidos.push(a);
      }
      return escolhidos.map(function(a) { return { x: a.x / s, y: a.y / s, lado: a.ws / s }; });
    }

    // Recorte quadrado centrado em (cx, cy); o que passar da borda da imagem fica branco
    function recortarCentrado(fonte, W, H, cx, cy, lado, saidaMax) {
      const esc = Math.min(1, saidaMax / lado);
      const c = document.createElement('canvas');
      c.width = c.height = Math.max(1, Math.round(lado * esc));
      const x = c.getContext('2d');
      x.fillStyle = '#fff'; x.fillRect(0, 0, c.width, c.height);
      x.imageSmoothingQuality = 'high';
      const sx0 = cx - lado / 2, sy0 = cy - lado / 2;
      const sx = Math.max(0, sx0), sy = Math.max(0, sy0);
      const sw = Math.min(W, sx0 + lado) - sx, sh = Math.min(H, sy0 + lado) - sy;
      if (sw > 0 && sh > 0) x.drawImage(fonte, sx, sy, sw, sh, (sx - sx0) * esc, (sy - sy0) * esc, sw * esc, sh * esc);
      return c;
    }

    // Recorte retangular do centro (fração da imagem), reduzido até "saidaMax"
    function recortarCentro(fonte, W, H, fracao, saidaMax) {
      const lw = Math.round(W * fracao), lh = Math.round(H * fracao);
      const esc = Math.min(1, saidaMax / Math.max(lw, lh));
      const c = document.createElement('canvas');
      c.width = Math.max(1, Math.round(lw * esc)); c.height = Math.max(1, Math.round(lh * esc));
      const x = c.getContext('2d');
      x.imageSmoothingQuality = 'high';
      x.drawImage(fonte, Math.round((W - lw) / 2), Math.round((H - lh) / 2), lw, lh, 0, 0, c.width, c.height);
      return c;
    }

    // Tenta várias leituras em sequência; para na primeira que der certo
    function tentarEmSequencia(geradores) {
      let i = 0;
      function proxima() {
        if (i >= geradores.length) return Promise.resolve(null);
        const canvas = geradores[i++]();
        return decodificar(canvas).then(function(t) { return t || proxima(); });
      }
      return proxima();
    }

    // Leitura completa de uma imagem (foto): inteira, centro e cada área com cara de código
    function lerCodigoEm(fonte, W, H) {
      const g = [
        function() { return recortarCentro(fonte, W, H, 1, detectorNativo ? 2000 : 1600); },
        function() { return recortarCentro(fonte, W, H, 0.6, 1600); }
      ];
      candidatosCodigo(fonte, W, H, 6).forEach(function(cd, i) {
        g.push(function() { return recortarCentrado(fonte, W, H, cd.x, cd.y, Math.max(160, cd.lado * 1.8), 900); });
        if (i < 2) g.push(function() { return recortarCentrado(fonte, W, H, cd.x, cd.y, Math.max(240, cd.lado * 3), 1000); });
      });
      return tentarEmSequencia(g);
    }

    // Lê um quadro da câmera. Alterna: quadrado grande do centro (75%), quadrado fechado (45%)
    // e, a cada 3 quadros, procura o código no quadro inteiro (não precisa estar centralizado).
    // Se demorar, mostra dica (no celular o mais comum é estar perto demais e a câmera não focar)
    let inicioLeituraQR = 0;
    function dicaQR(demorando) {
      const el = document.getElementById('qrDica');
      el.className = demorando ? 'small mt-2 text-warning-emphasis fw-semibold' : 'small text-muted mt-2';
      el.innerText = demorando
        ? 'Ainda procurando o código... Afaste um pouco o celular (uns 15 cm) para a câmera focar, deixe o código dentro do quadrado e com boa luz. Se tiver, use o zoom.'
        : 'Aponte a câmera para o código da etiqueta, dentro do quadrado. A leitura é automática — não precisa tirar foto.';
    }

    function lerQuadro() {
      if (!lendoQR) return;
      if (inicioLeituraQR && Date.now() - inicioLeituraQR > 6000) { dicaQR(true); inicioLeituraQR = 0; }
      const vw = videoQR.videoWidth, vh = videoQR.videoHeight;
      if (!vw || !vh) { setTimeout(lerQuadro, 200); return; }
      const etapa = cicloQR++ % 3;
      let leitura;
      if (etapa === 2) {
        // procura o código no quadro inteiro e lê centrado nele
        const quadro = recortarCentro(videoQR, vw, vh, 1, 1280);
        const qw = quadro.width, qh = quadro.height;
        const g = detectorNativo
          ? [function() { return quadro; }]
          : candidatosCodigo(quadro, qw, qh, 3).map(function(cd) {
              return function() { return recortarCentrado(quadro, qw, qh, cd.x, cd.y, Math.max(120, cd.lado * 1.8), 700); };
            });
        leitura = tentarEmSequencia(g);
      } else {
        const fracao = etapa === 0 ? 0.75 : 0.45;
        const lado = Math.round(Math.min(vw, vh) * fracao);
        // resolução real da câmera (limitada a 1200 px)
        leitura = decodificar(recortarCentrado(videoQR, vw, vh, vw / 2, vh / 2, lado, 1200));
      }
      leitura.then(function(texto) {
        if (!lendoQR) return;
        if (texto) {
          if (parsearEtiqueta(texto)) {
            lendoQR = false;
            document.getElementById('qrMira').classList.add('lido');
            if (navigator.vibrate) navigator.vibrate(80);
            aplicarQR(texto);
            return;
          }
          aplicarQR(texto); // código lido, mas de outro formato: mostra o aviso e continua
        }
        setTimeout(lerQuadro, 60);
      }).catch(function() { if (lendoQR) setTimeout(lerQuadro, 150); });
    }

    function pararLeitor() {
      lendoQR = false;
      sessaoQR++;
      pararStreamQR();
      return Promise.resolve();
    }

    // Zoom e lanterna, quando a câmera do aparelho oferece
    function configurarZoomLanterna(track) {
      const caixa = document.getElementById('qrControles');
      const zoomBox = document.getElementById('qrZoomBox');
      const lanterna = document.getElementById('qrLanterna');
      caixa.classList.add('d-none'); zoomBox.classList.add('d-none'); lanterna.classList.add('d-none');
      if (!track || !track.getCapabilities) return;
      let cap = {};
      try { cap = track.getCapabilities() || {}; } catch (e) { return; }
      // foco contínuo, se der
      if (cap.focusMode && cap.focusMode.indexOf('continuous') >= 0) {
        track.applyConstraints({ advanced: [{ focusMode: 'continuous' }] }).catch(function() {});
      }
      if (cap.zoom && cap.zoom.max > cap.zoom.min) {
        const r = document.getElementById('qrZoom');
        const atual = (track.getSettings && track.getSettings().zoom) || cap.zoom.min;
        r.min = cap.zoom.min; r.max = cap.zoom.max; r.step = cap.zoom.step || 0.1; r.value = atual;
        document.getElementById('qrZoomVal').innerText = (+atual).toFixed(1) + 'x';
        r.oninput = function() {
          document.getElementById('qrZoomVal').innerText = (+r.value).toFixed(1) + 'x';
          track.applyConstraints({ advanced: [{ zoom: +r.value }] }).catch(function() {});
        };
        zoomBox.classList.remove('d-none');
        caixa.classList.remove('d-none');
      }
      if (cap.torch) {
        let ligada = false;
        lanterna.onclick = function() {
          ligada = !ligada;
          track.applyConstraints({ advanced: [{ torch: ligada }] }).catch(function() {});
          lanterna.classList.toggle('active', ligada);
        };
        lanterna.classList.remove('d-none');
        caixa.classList.remove('d-none');
      }
    }

    // Lê o código de uma foto (procura o código na foto inteira — não precisa estar centralizado)
    function lerCodigoDeArquivo(file) {
      return new Promise(function(ok, falha) {
        const url = URL.createObjectURL(file);
        const img = new Image();
        img.onload = function() { URL.revokeObjectURL(url); ok(img); };
        img.onerror = function() { URL.revokeObjectURL(url); falha(new Error('imagem inválida')); };
        img.src = url;
      }).then(function(img) {
        return prepararDecodificador().then(function() {
          return lerCodigoEm(img, img.naturalWidth, img.naturalHeight);
        });
      });
    }

    function enviarFormulario(e) {
      e.preventDefault();
      const btn = document.getElementById('btnSalvar');
      const alerta = document.getElementById('alerta');

      if (!base64Imagem) {
        alerta.className = 'alert alert-warning';
        alerta.innerText = 'Tire ou escolha a foto da PCB primeiro (Passo 1).';
        alerta.classList.remove('d-none');
        window.scrollTo({ top: 0, behavior: 'smooth' });
        return;
      }

      btn.disabled = true;
      btn.innerHTML = '<span class="spinner-border spinner-border-sm me-2"></span>Salvando...';

      const payload = {
        codigo: formatarCodigos(document.getElementById('codigo').value, false),
        quantPainel: document.getElementById('quantPainel').value,
        quantPacote: valorPacote(),
        fornecedor: document.getElementById('fornecedor').value,
        imagemBase64: base64Imagem
      };

      google.script.run
        .withSuccessHandler(function(res) {
          btn.disabled = false;
          btn.innerHTML = '<i class="bi bi-check-lg me-1"></i> Salvar Registro';
          alerta.className = 'alert ' + (res.sucesso ? 'alert-success' : 'alert-danger');
          alerta.innerText = res.mensagem;
          alerta.classList.remove('d-none');

          if (res.sucesso) {
            document.getElementById('formCadastro').reset();
            removerFoto();
            Object.keys(quantGuardadas).forEach(function(k) { delete quantGuardadas[k]; });
            atualizarQuantPorCodigo();
            document.getElementById('qrStatus').innerText = '';
            carregarFornecedores();
          }
        })
        .withFailureHandler(function(err) {
          btn.disabled = false;
          btn.innerHTML = '<i class="bi bi-check-lg me-1"></i> Salvar Registro';
          alerta.className = 'alert alert-danger';
          alerta.innerText = 'Falha ao salvar: ' + err.message;
          alerta.classList.remove('d-none');
        })
        .salvarRegistro(payload);
    }

    function carregarDados() {
      const corpo = document.getElementById('tabelaCorpo');
      corpo.innerHTML = '<tr><td colspan="5" class="text-center py-4"><span class="spinner-border spinner-border-sm me-2"></span>Buscando PCBs na planilha...</td></tr>';

      google.script.run
        .withSuccessHandler(function(dados) {
          registrosGlobais = dados;
          atualizarFornecedores(dados);
          renderizarTabela(dados);
        })
        .withFailureHandler(function(err) {
          corpo.innerHTML = `<tr><td colspan="5" class="text-center text-danger">Erro: ${err.message}</td></tr>`;
        })

    }

    // ===== Paginação da lista (10 por página) =====
    const POR_PAGINA = 10;
    let listaExibida = [];
    let paginaAtual = 1;

    function renderizarPaginacao() {
      const total = Math.ceil(listaExibida.length / POR_PAGINA);
      const ul = document.getElementById('paginacao');
      const info = document.getElementById('infoPaginacao');

      if (listaExibida.length === 0) {
        ul.innerHTML = '';
        info.innerText = '';
        return;
      }

      const ini = (paginaAtual - 1) * POR_PAGINA + 1;
      const fim = Math.min(paginaAtual * POR_PAGINA, listaExibida.length);
      info.innerText = 'Mostrando ' + ini + '–' + fim + ' de ' + listaExibida.length + ' · Página ' + paginaAtual + ' de ' + total;

      const item = function(rotulo, pagina, desativado, ativo) {
        return '<li class="page-item' + (desativado ? ' disabled' : '') + (ativo ? ' active' : '') + '">' +
          '<button type="button" class="page-link" data-pagina="' + pagina + '"' + (desativado ? ' disabled' : '') + '>' + rotulo + '</button></li>';
      };

      let html = item('Anterior', paginaAtual - 1, paginaAtual === 1, false);
      let ultimoFoiReticencias = false;
      for (let i = 1; i <= total; i++) {
        if (i === 1 || i === total || Math.abs(i - paginaAtual) <= 1) {
          html += item(i, i, false, i === paginaAtual);
          ultimoFoiReticencias = false;
        } else if (!ultimoFoiReticencias) {
          html += '<li class="page-item disabled"><span class="page-link">…</span></li>';
          ultimoFoiReticencias = true;
        }
      }
      html += item('Próximo', paginaAtual + 1, paginaAtual === total, false);
      ul.innerHTML = html;
    }

    document.getElementById('paginacao').addEventListener('click', function(e) {
      const botao = e.target.closest('button[data-pagina]');
      if (!botao || botao.disabled) return;
      renderizarTabela(listaExibida, parseInt(botao.dataset.pagina, 10));
      document.querySelector('.lista-scroll').scrollTop = 0;
    });

    function renderizarTabela(lista, pagina) {
      const corpo = document.getElementById('tabelaCorpo');
      listaExibida = lista || [];
      const totalPaginas = Math.max(1, Math.ceil(listaExibida.length / POR_PAGINA));
      paginaAtual = Math.min(Math.max(pagina || 1, 1), totalPaginas);

      if (listaExibida.length === 0) {
        corpo.innerHTML = '<tr><td colspan="5" class="text-center text-muted">Nenhum registro encontrado.</td></tr>';
        renderizarPaginacao();
        return;
      }
      const itensDaPagina = listaExibida.slice((paginaAtual - 1) * POR_PAGINA, paginaAtual * POR_PAGINA);

      const esc = v => String(v == null ? '' : v).replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));

      // "5000 / 3000" com 2 códigos -> uma linha por código; senão mostra o valor como está
      const formatarPacote = (item, esc) => {
        const q = String(item.quantPacote == null ? '' : item.quantPacote).split(/\s*\/\s*/).filter(Boolean);
        const c = String(item.codigo).split(/\s*\/\s*/).filter(Boolean);
        if (q.length > 1 && q.length === c.length) {
          return c.map((cod, i) => `<div><span class="text-muted small">${esc(cod)}</span> <b>${esc(q[i])}</b></div>`).join('');
        }
        return esc(item.quantPacote);
      };

      corpo.innerHTML = itensDaPagina.map(item => `
        <tr>
          <td>
            ${item.imagem
              ? `<img src="${esc(item.imagem)}" data-full="${esc(item.imagem)}" data-legenda="${esc(item.codigo)}" data-linha="${esc(item.linhaIndex)}" data-coluna="${esc(item.colunaImagem)}" title="Clique para ampliar" class="table-img border" onerror="this.src='https://via.placeholder.com/65?text=Erro'">`
              : '<span class="text-muted small">Sem foto</span>'}
          </td>
          <td class="col-codigo">${esc(item.codigo)}</td>
          <td class="col-num">${esc(item.quantPainel)}</td>
          <td>${esc(item.fornecedor)}</td>
          <td class="col-num">${formatarPacote(item, esc)}</td>
        </tr>
      `).join('');
      renderizarPaginacao();
    }

    // ===== Ajustar uma foto já cadastrada (recortar, clarear, nitidez) =====
    let fotoAberta = null; // { url, linha, coluna } da foto exibida na janela ampliada

          erro.innerText = 'Não foi possível carregar a foto: ' + err.message;
          erro.classList.remove('d-none');

    }

      const btn = document.getElementById('btnUsarFoto');

    // Clique na miniatura da tabela -> foto ampliada (com zoom)
    const elModalFoto = document.getElementById('modalFoto');
    const modalFoto = new bootstrap.Modal(elModalFoto);
    const viewportFoto = document.getElementById('fotoViewport');
    const imgFoto = document.getElementById('fotoGrande');
    const ZOOM_MAX = 10;
    let zoom = 1, desX = 0, desY = 0;
    const ponteiros = new Map();
    let distanciaPinca = 0;

    document.getElementById('tabelaCorpo').addEventListener('click', function(e) {
      const img = e.target.closest('img[data-full]');
      if (!img) return;
      resetarZoomFoto();
      const aberta = { url: img.dataset.full, linha: parseInt(img.dataset.linha, 10), coluna: parseInt(img.dataset.coluna, 10), dataUrl: null };
      fotoAberta = aberta;
      document.getElementById('fotoErro').classList.add('d-none');
      imgFoto.src = img.dataset.full;
      // o link lh3 do Google pode vir reduzido/recomprimido: troca pelo arquivo salvo no Drive
      google.script.run
        .withSuccessHandler(function(dataUrl) {
          aberta.dataUrl = dataUrl;
          if (fotoAberta === aberta) imgFoto.src = dataUrl;
        })
        .withFailureHandler(function() {})
        .obterFotoBase64(aberta.url);
      document.getElementById('fotoAbrirAba').href = img.dataset.full;
      document.getElementById('fotoLegenda').innerText = 'Código ' + (img.dataset.legenda || '');
      modalFoto.show();
    });

    function aplicarZoomFoto() {
      imgFoto.style.transform = 'translate(' + desX + 'px,' + desY + 'px) scale(' + zoom + ')';
      viewportFoto.classList.toggle('ampliada', zoom > 1);
      document.getElementById('fotoZoomInfo').innerText = Math.round(zoom * 100) + '%';
    }

    function resetarZoomFoto() {
      zoom = 1; desX = 0; desY = 0;
      aplicarZoomFoto();
    }

    // Zoom mantendo fixo o ponto sob o cursor. (cx, cy) em px a partir do centro da área.
    function zoomFotoEm(fator, cx, cy) {
      const novo = Math.min(ZOOM_MAX, Math.max(1, zoom * fator));
      if (novo === zoom) return;
      const ux = (cx - desX) / zoom, uy = (cy - desY) / zoom;
      zoom = novo;
      desX = cx - ux * zoom;
      desY = cy - uy * zoom;
      if (zoom === 1) { desX = 0; desY = 0; }
      aplicarZoomFoto();
    }

    function zoomFoto(fator) { zoomFotoEm(fator, 0, 0); }

    function centroRelativo(clientX, clientY) {
      const r = viewportFoto.getBoundingClientRect();
      return { x: clientX - (r.left + r.width / 2), y: clientY - (r.top + r.height / 2) };
    }

    // Roda do mouse
    viewportFoto.addEventListener('wheel', function(e) {
      e.preventDefault();
      const p = centroRelativo(e.clientX, e.clientY);
      zoomFotoEm(e.deltaY < 0 ? 1.2 : 1 / 1.2, p.x, p.y);
    }, { passive: false });

    // Duplo clique / duplo toque: amplia no ponto clicado ou volta ao ajuste.
    // Detectado à mão (dois toques curtos seguidos) porque o evento dblclick não dispara em celular.
    let inicioToque = null;
    let ultimoToque = { t: 0, x: 0, y: 0 };

    function duploToque(clientX, clientY) {
      if (zoom > 1) { resetarZoomFoto(); return; }
      const p = centroRelativo(clientX, clientY);
      zoomFotoEm(3, p.x, p.y);
    }

    // Arrastar para mover (mouse e toque) e pinça com dois dedos
    viewportFoto.addEventListener('pointerdown', function(e) {
      viewportFoto.setPointerCapture(e.pointerId);
      ponteiros.set(e.pointerId, { x: e.clientX, y: e.clientY });
      inicioToque = ponteiros.size === 1 ? { x: e.clientX, y: e.clientY } : null;
      if (ponteiros.size === 2) {
        const [a, b] = Array.from(ponteiros.values());
        distanciaPinca = Math.hypot(a.x - b.x, a.y - b.y);
      }
      if (zoom > 1) viewportFoto.classList.add('arrastando');
    });

    viewportFoto.addEventListener('pointermove', function(e) {
      const anterior = ponteiros.get(e.pointerId);
      if (!anterior) return;
      ponteiros.set(e.pointerId, { x: e.clientX, y: e.clientY });

      if (ponteiros.size === 2) {
        const [a, b] = Array.from(ponteiros.values());
        const dist = Math.hypot(a.x - b.x, a.y - b.y);
        if (distanciaPinca) {
          const p = centroRelativo((a.x + b.x) / 2, (a.y + b.y) / 2);
          zoomFotoEm(dist / distanciaPinca, p.x, p.y);
        }
        distanciaPinca = dist;
      } else if (zoom > 1) {
        desX += e.clientX - anterior.x;
        desY += e.clientY - anterior.y;
        aplicarZoomFoto();
      }
    });

    function soltarPonteiro(e) {
      ponteiros.delete(e.pointerId);
      distanciaPinca = 0;
      if (ponteiros.size === 0) viewportFoto.classList.remove('arrastando');
    }
    viewportFoto.addEventListener('pointerup', function(e) {
      // toque curto, sem arrastar? então conta para o duplo toque
      if (inicioToque && ponteiros.size === 1 && Math.hypot(e.clientX - inicioToque.x, e.clientY - inicioToque.y) < 8) {
        const agora = Date.now();
        if (agora - ultimoToque.t < 350 && Math.hypot(e.clientX - ultimoToque.x, e.clientY - ultimoToque.y) < 30) {
          ultimoToque.t = 0;
          duploToque(e.clientX, e.clientY);
        } else {
          ultimoToque = { t: agora, x: e.clientX, y: e.clientY };
        }
      }
      inicioToque = null;
      soltarPonteiro(e);
    });
    viewportFoto.addEventListener('pointercancel', function(e) { inicioToque = null; soltarPonteiro(e); });

    function limparBusca() {
      const campo = document.getElementById('filtroPesquisa');
      campo.value = '';
      filtrarTabela();
      campo.focus();
    }

    function filtrarTabela() {
      const termo = document.getElementById('filtroPesquisa').value.toLowerCase();
      document.getElementById('btnLimparBusca').classList.toggle('d-none', termo === '');
      const filtrados = registrosGlobais.filter(item => 
        (item.codigo && item.codigo.toString().toLowerCase().includes(termo)) ||
        (item.fornecedor && item.fornecedor.toString().toLowerCase().includes(termo))
      );
      renderizarTabela(filtrados);
    }
